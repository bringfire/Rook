# OCCT build recipe for the DLLs Rook ships next to RookNative (#598).
#
# RECONSTRUCTED on 2026-09-26 from the CMakeCache.txt and CMakeCXXCompiler.cmake of
# the build that produced the shipped DLLs (their sha256 values are pinned in
# third_party/occt/occt-provenance.json). It records what that build used; it was
# not the original command line.
#
# Source:    https://github.com/Open-Cascade-SAS/OCCT
#            tag V8_0_0 = commit d3056ef80c9668f395da40f5fd7be186cae4501f, unmodified
# Toolchain: Visual Studio 2022 17.8, MSVC toolset 14.38.33130 (cl 19.38.33145.0), x64
# Output:    <build>\win64\vc14\bin (the 11 DLLs listed in occt-provenance.json)

cmake -S <OCCT> -B <OCCT>\build-rook -G "Visual Studio 17 2022" -A x64 ^
  -DBUILD_LIBRARY_TYPE=Shared ^
  -DBUILD_CPP_STANDARD=C++17 ^
  -DBUILD_MODULE_FoundationClasses=ON ^
  -DBUILD_MODULE_ModelingData=ON ^
  -DBUILD_MODULE_ModelingAlgorithms=ON ^
  -DBUILD_MODULE_DataExchange=ON ^
  -DBUILD_MODULE_ApplicationFramework=OFF ^
  -DBUILD_MODULE_Visualization=OFF ^
  -DBUILD_MODULE_Draw=OFF ^
  -DBUILD_MODULE_DETools=OFF ^
  -DBUILD_DOC_Overview=OFF ^
  -DBUILD_GTEST=OFF ^
  -DBUILD_RELEASE_DISABLE_EXCEPTIONS=ON ^
  -DBUILD_USE_PCH=OFF ^
  -DBUILD_USE_VCPKG=OFF ^
  -DBUILD_OPT_PROFILE=Default ^
  -DUSE_MMGR_TYPE=NATIVE ^
  -DUSE_TBB=OFF -DUSE_FREETYPE=OFF -DUSE_FREEIMAGE=OFF -DUSE_FFMPEG=OFF ^
  -DUSE_RAPIDJSON=OFF -DUSE_DRACO=OFF -DUSE_OPENVR=OFF -DUSE_D3D=OFF ^
  -DINSTALL_DIR_LAYOUT=Windows

cmake --build <OCCT>\build-rook --config Release

# Effective Release flags: CMAKE_CXX_FLAGS "/DWIN32 /D_WINDOWS /W3 /GR /EHsc",
# CMAKE_CXX_FLAGS_RELEASE "/MD /O2 /Ob2 /DNDEBUG". DataExchange was built but none
# of its toolkits ship: RookNative's runtime closure is the 11 modeling DLLs only.
