# PR-18 Seedance 2.0 Image-to-Video Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add curated fal.ai `bytedance/seedance-2.0/image-to-video` as a source-image video model using artifact-id source frames, model-aware fal video lifecycle calls, request-id-only durable Seedance handles, and local `generated_video` materialization.

**Architecture:** Keep provider identity as `fal` and make `FalVideoProvider` model-aware: existing Wan T2V remains URL-handle based, while Seedance I2V/Interp reconstructs fal status/result/cancel endpoints from explicit `modelId + request_id`. Add an explicit video lifecycle contract (`IModelAwareVideoProvider`) and have `VideoJobManager` pass resolved model identity during active polling/fetch/cancel and validated durable model identity during restart cancel. Source image bytes are resolved from artifact-id refs only, validated as bounded PNG/JPEG/WebP, sent as data URIs, and never persisted.

**Tech Stack:** C#/.NET managed Rook Vision stack, existing `FalApiClient`, existing video job manager/ledger/materializer, Vision WebView JavaScript, xUnit fake HTTP tests.

public Task<ProviderStatusOutcome> GetStatusAsync(
    string modelId,
    ProviderJobHandle handle,
    CancellationToken ct)
{
    RecordedCalls.Add(("GetStatusForModel", new
    {
        modelId,
        providerJobId = handle.ProviderJobId,
    }));
    var result = OnGetStatusForModel?.Invoke(modelId, handle)
        ?? OnGetStatus?.Invoke(handle)
        ?? StatusInFlight(50, "polling");
    return Task.FromResult(result);
}

public Task<ProviderCancelOutcome> CancelAsync(
    string modelId,
    ProviderJobHandle handle,
    CancellationToken ct)
{
    RecordedCalls.Add(("CancelForModel", new
    {
        modelId,
        providerJobId = handle.ProviderJobId,
    }));
    var result = OnCancelForModel?.Invoke(modelId, handle)
        ?? OnCancel?.Invoke(handle)
        ?? CancelOk();
    return Task.FromResult(result);
}

public Task<ProviderResultOutcome> FetchResultAsync(
    string modelId,
    ProviderJobHandle handle,
    CancellationToken ct)
{
    RecordedCalls.Add(("FetchResultForModel", new
    {
        modelId,
        providerJobId = handle.ProviderJobId,
        providerResultToken = handle.ProviderResultToken,
    }));
    var result = OnFetchResultForModel?.Invoke(modelId, handle)
        ?? OnFetchResult?.Invoke(handle)
        ?? ResultFailed(new VideoJobError(
            Code: VideoErrorCode.ExecutionFailed,
            Message: "Fetch called before completion (default fake behaviour).",
            Retryable: true));
    return Task.FromResult(result);
}

Add this helper next to `TryRemoteCancelAsync`:

```csharp
private static async Task<JobCancelResult> TryLegacyRemoteCancelAsync(
    IVideoProvider provider,
    ProviderJobHandle handle,
    CancellationToken ct)
{
    try
    {
        var outcome = await provider.CancelAsync(handle, ct)
            .ConfigureAwait(false);
        return outcome switch
        {
            CanceledOutcome => JobCancelResult.Ok(VideoJobState.Cancelled),
            AlreadyTerminalOutcome already => JobCancelResult.Ok(
                already.State == GenerationLifecycleState.Completed
                    ? VideoJobState.Complete
                    : VideoJobState.Error),
            FailedCancelOutcome failed => JobCancelResult.Fail(
                VideoProviderOutcomeAdapters.ToVideoJobError(failed.Error)),
            _ => JobCancelResult.Fail(new VideoJobError(
                VideoErrorCode.ExecutionFailed,
                $"Unknown provider cancel outcome: {outcome.GetType().Name}.",
                Retryable: false)),
        };
    }
    catch (OperationCanceledException) when (ct.IsCancellationRequested)
    {
        throw;
    }
    catch (Exception ex)
    {
        return JobCancelResult.Fail(new VideoJobError(
            VideoErrorCode.DependencyUnavailable,
            $"Provider cancel failed: {ex.Message}",
            Retryable: true));
    }
}
```

using System;
using System.Collections.Generic;
using Rook.Services.Vision.Generation;
using Rook.Services.Vision.Image;

namespace Rook.Services.Vision.Video.Fal
{
    internal sealed class FalSeedanceI2vSourcePayload
    {
        public const long MaxRawBytes = 1024 * 1024;

        private FalSeedanceI2vSourcePayload(string imageUrl, string? endImageUrl)
        {
            ImageUrl = imageUrl;
            EndImageUrl = endImageUrl;
        }

        public string ImageUrl { get; }
        public string? EndImageUrl { get; }

        public static (
            FalSeedanceI2vSourcePayload? Payload,
            GenerationError? Error) FromResolvedMedia(
            VideoGenerationRequest request,
            IReadOnlyDictionary<MediaRef, ResolvedMedia> media)
        {
            if (request is null)
                return (null, Invalid("Request is null.", nameof(request)));

            if (media is null)
                return (null, Invalid(
                    "Seedance 2.0 Image to Video requires source image media.",
                    nameof(VideoGenerationRequest.StartFrame)));

            if (request.ReferenceFrames is { Count: > 0 })
                return (null, Invalid(
                    "Seedance 2.0 Image to Video does not support reference frames in PR-18.",
                    nameof(VideoGenerationRequest.ReferenceFrames)));

            if (request.Mode != VideoMode.I2V && request.Mode != VideoMode.Interp)
                return (null, Invalid(
                    "Seedance 2.0 Image to Video supports only I2V and Interp in PR-18.",
                    nameof(VideoGenerationRequest.Mode)));

            if (request.StartFrame is null)
                return (null, Invalid(
                    "Seedance 2.0 Image to Video requires a start frame.",
                    nameof(VideoGenerationRequest.StartFrame)));

            var (startUri, startError) = BuildDataUri(
                request.StartFrame,
                media,
                nameof(VideoGenerationRequest.StartFrame));
            if (startError is not null)
                return (null, startError);

            string? endUri = null;
            if (request.Mode == VideoMode.Interp)
            {
                if (request.EndFrame is null)
                    return (null, Invalid(
                        "Seedance 2.0 interpolation requires an end frame.",
                        nameof(VideoGenerationRequest.EndFrame)));

                var (builtEnd, endError) = BuildDataUri(
                    request.EndFrame,
                    media,
                    nameof(VideoGenerationRequest.EndFrame));
                if (endError is not null)
                    return (null, endError);
                endUri = builtEnd;
            }
            else if (request.EndFrame is not null)
            {
                return (null, Invalid(
                    "Seedance 2.0 I2V does not accept an end frame.",
                    nameof(VideoGenerationRequest.EndFrame)));
            }

            return (new FalSeedanceI2vSourcePayload(startUri!, endUri), null);
        }

        private static (string? DataUri, GenerationError? Error) BuildDataUri(
            MediaRef mediaRef,
            IReadOnlyDictionary<MediaRef, ResolvedMedia> media,
            string field)
        {
            if (!media.TryGetValue(mediaRef, out var resolved))
                return (null, Invalid(
                    "Seedance 2.0 source frame bytes were not resolved.",
                    field));

            if (resolved.Bytes is null || resolved.Bytes.Length == 0)
                return (null, Invalid(
                    "Seedance 2.0 source frame is empty.",
                    field));

            if (resolved.Bytes.LongLength > MaxRawBytes)
                return (null, Invalid(
                    "Seedance 2.0 source frame is too large for data URI upload; " +
                    "capture a smaller viewport or lower resolution.",
                    field));

            var detectedMime = ImageMimeDetector.Detect(resolved.Bytes);
            if (!ImageMimeDetector.IsPngJpegOrWebp(detectedMime))
                return (null, Invalid(
                    "Seedance 2.0 source frame must be PNG, JPEG, or WebP.",
                    field));

            if (!string.IsNullOrWhiteSpace(resolved.MimeType)
                && !string.Equals(resolved.MimeType, detectedMime, StringComparison.Ordinal))
            {
                return (null, Invalid(
                    "Seedance 2.0 source frame MIME does not match its bytes.",
                    field));
            }

            return (
                $"data:{detectedMime};base64,{Convert.ToBase64String(resolved.Bytes)}",
                null);
        }

        private static GenerationError Invalid(string message, string field) =>
            new(
                Code: GenerationErrorCode.InvalidRequest,
                Message: message,
                Retryable: false,
                Field: field);
    }
}

            GenerationErrorCode.InvalidRequest,
            "Seedance 2.0 Image to Video requires prompt.",
            nameof(VideoGenerationRequest.Prompt));

        return FailedSubmit(
            GenerationErrorCode.InvalidRequest,
            "Seedance 2.0 Image to Video supports exactly one video in PR-18.",
            nameof(VideoGenerationRequest.NumberOfVideos));

        FalSeedanceI2vSourcePayload.FromResolvedMedia(request, resolvedMedia);
    if (sourceError is not null)
        return new FailedSubmitOutcome(sourceError);

    var apiKey = _apiKeyProvider();
    if (string.IsNullOrWhiteSpace(apiKey))
        return FailedSubmit(
            GenerationErrorCode.DependencyUnavailable,
            "fal API key is not configured.");

    FalHttpResponse response;
    try
    {
        response = await _client.PostJsonAsync(
            apiKey!,
            SeedanceEndpoint,
            BuildSeedanceRequestJson(request, source!),
            ct).ConfigureAwait(false);
    }
    catch (OperationCanceledException) when (ct.IsCancellationRequested)
    {
        throw;
    }
    catch (TaskCanceledException)
    {
        return FailedSubmit(
            GenerationErrorCode.DependencyUnavailable,
            "fal Seedance submit timed out.",
            retryable: true);
    }
    catch (HttpRequestException)
    {
        return FailedSubmit(
            GenerationErrorCode.DependencyUnavailable,
            "fal Seedance submit failed due to a transport error.",
            retryable: true);
    }

    if (!response.IsSuccessStatusCode)
        return new FailedSubmitOutcome(FalErrorMapper.MapHttpFailure(response));

    try
    {
        var root = JsonNode.Parse(response.Body) as JsonObject;
        if (root is null)
            return FailedSubmit(
                GenerationErrorCode.ExecutionFailed,
                "fal Seedance submit response was empty.");

        if (!TryGetString(root, "request_id", out var requestId))
            return FailedSubmit(
                GenerationErrorCode.ExecutionFailed,
                "fal Seedance submit response did not contain request_id.");

        return new QueuedSubmitOutcome(new ProviderJobHandle(requestId!));
    }
    catch (JsonException)
    {
        return FailedSubmit(
            GenerationErrorCode.ExecutionFailed,
            "fal Seedance submit response was not valid JSON.");
    }
    catch (ArgumentException ex)
    {
        return FailedSubmit(
            GenerationErrorCode.ExecutionFailed,
            ex.Message);
    }
}

private static ProviderResultOutcome ParseSeedanceFetchResult(string body)
{
    JsonObject root;
    try
    {
        root = JsonNode.Parse(body) as JsonObject
            ?? throw new JsonException();
    }
    catch (JsonException)
    {
        return FailedResult(
            GenerationErrorCode.ExecutionFailed,
            "fal Seedance result response was not valid JSON.");
    }

    if (root["video"] is not JsonObject video)
        return FailedResult(
            GenerationErrorCode.ExecutionFailed,
            "fal Seedance result response did not contain video object.",
            "video");

    if (!TryGetString(video, "url", out var urlText)
        || !Uri.TryCreate(urlText, UriKind.Absolute, out var url)
        || (url.Scheme != Uri.UriSchemeHttp && url.Scheme != Uri.UriSchemeHttps))
    {
        return FailedResult(
            GenerationErrorCode.ExecutionFailed,
            "fal Seedance video URL was missing or invalid.",
            "video.url");
    }

    var artifact = new ResultArtifact(
        Role: VideoMediaRoles.Video,
        Body: new RemoteArtifactBody(url),
        DeclaredMimeType: TryGetString(video, "content_type", out var mime)
            ? mime
            : null,
        ProviderMetadata: EmptyMetadata);

    return new SuccessResultOutcome(
        new ProviderResultEnvelope(
            new[] { artifact },
            EmptyMetadata));
